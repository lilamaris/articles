## 내 접근
- `1 <= N <= 5 x 10^5` 조건에서 행렬의 모든 값을 다 채우고 카운팅은 메모리 제약 조건을 쉽게 초과한다.  현재 좌표의 값은 이전 행/열 값을 참조하니까 N 크기의 두 배열만 두면 메모리 제약을 해소할 수 있을 것
- 그럼에도 시간 복잡도는 `O(n^2)`이라 시간 제약을 통과하긴 어렵다

```cpp
#include <iostream>
#define IO std::cin.tie(NULL), std::ios_base::sync_with_stdio(false)

#define MAX 500000
int first[MAX];
int second[MAX];

int mex(int i, int j) {
  bool exists[3] = {
      false,
  };
  exists[i] = true;
  exists[j] = true;
  for (int i = 0; i < 3; i++) {
    if (!exists[i])
      return i;
  }
  return 0;
}

void solve() {
  int T, N;
  std::cin >> T;

  while (T--) {
    int count[3];

    std::cin >> N;

    std::fill(count, count + 3, 0);

    int *previous = first;
    int *current = second;

    for (int col = 0; col < N; col++) {
      int x;
      std::cin >> x;
      count[x]++;
      previous[col] = x;
    }

    for (int row = 1; row < N; row++) {
      int x;
      std::cin >> x;
      current[0] = x;
      count[x]++;

      for (int col = 1; col < N; col++) {
        int x = mex(current[col - 1], previous[col]);
        count[x]++;
        current[col] = x;
      }

      int *tmp = previous;
      previous = current;
      current = tmp;
    }

    for (int i = 0; i < 3; i++) {
      std::cout << count[i] << " ";
    }
    std::cout << '\n';
  }
}

int main() {
  IO;
  solve();
  return 0;
}
```
